    70→    70→function priorityOf(severity: string): { label: string; chip: string } {
    71→    71→  switch (severity) {
    72→    72→    case "Critical":
    73→    73→      return { label: "P1", chip: "border-red-500/40 bg-red-500/10 text-red-300" };
    74→    74→    case "High":
    75→    75→      return { label: "P2", chip: "border-amber-500/40 bg-amber-500/10 text-amber-300" };
    76→    76→    case "Medium":
    77→    77→      return { label: "P3", chip: "border-yellow-400/40 bg-yellow-400/10 text-yellow-300" };
    78→    78→    case "Low":
    79→    79→      return { label: "P4", chip: "border-emerald-400/40 bg-emerald-400/10 text-emerald-300" };
    80→    80→    default:
    81→    81→      return { label: "FP", chip: "border-slate-500/40 bg-slate-500/10 text-slate-300" };
    82→    82→  }

   356→   356→function IncidentTable({ incidents }: { incidents: IncidentDTO[] }) {
   357→   357→  const openIncident = useSocStore((s) => s.openIncident);
   358→   358→  const [exporting, setExporting] = useState(false);
   359→   359→  const [exportingPdf, setExportingPdf] = useState(false);
   360→   360→
   361→   361→  const fetchAllIncidents = async (): Promise<IncidentDTO[]> => {
   362→   362→    const all = await apiGet<unknown>("/api/incidents");
   363→   363→    return Array.isArray(all)
   364→   364→      ? (all as IncidentDTO[])
   365→   365→      : ((all as { incidents?: IncidentDTO[] }).incidents ?? []);
   366→   366→  };
   367→   367→
   368→   368→  const exportBriefing = async () => {
   369→   369→    setExportingPdf(true);
   370→   370→    try {
   371→   371→      const list = await fetchAllIncidents();
   372→   372→      if (list.length === 0) {
   373→   373→        toast.info("Nothing to brief — no incidents yet.");
   374→   374→        return;
   375→   375→      }
   376→   376→      const summary = await apiGet<DashboardSummary>("/api/dashboard/summary");
   377→   377→      const { downloadBriefingPackPdf } = await import("@/lib/report-pdf");
   378→   378→      await downloadBriefingPackPdf(list, {
   379→   379→        totalAlerts: summary.totalAlerts,
   380→   380→        unacknowledgedAlerts: summary.counts.unacknowledgedAlerts,
   381→   381→      });
   382→   382→      toast.success(`Briefing pack exported — ${list.length} incidents`);
   383→   383→    } catch (err) {
   384→   384→      toast.error("Briefing export failed", {
   385→   385→        description: err instanceof Error ? err.message : undefined,
   386→   386→      });
   387→   387→    } finally {
   388→   388→      setExportingPdf(false);
   389→   389→    }
   390→   390→  };
   391→   391→
   392→   392→  const exportCsv = async () => {
   393→   393→    setExporting(true);
   394→   394→    try {
   395→   395→      const list = await fetchAllIncidents();
   396→   396→      if (list.length === 0) {
   397→   397→        toast.info("Nothing to export — no incidents yet.");
   398→   398→        return;
   399→   399→      }
   400→   400→      const esc = (v: unknown) => {
   401→   401→        const s = String(v ?? "");
   402→   402→        return /["\n,]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
   403→   403→      };
   404→   404→      const header = [
   405→   405→        "incident_id", "title", "severity", "classification", "threat_score",
   406→   406→        "confidence_pct", "status", "alert_count", "sources", "ai_analyzed",
   407→   407→        "created_at", "updated_at",
   408→   408→      ].join(",");
   409→   409→      const rows = list.map((i) =>
   410→   410→        [
   411→   411→          i.incidentId, i.title, i.severity, i.classification, i.threatScore,
   412→   412→          i.confidence, i.status, i.alertCount, i.sources.join("; "), i.analyzed ? "yes" : "no",
   413→   413→          i.createdAt, i.updatedAt,
   414→   414→        ].map(esc).join(",")
   415→   415→      );
   416→   416→      const blob = new Blob([[header, ...rows].join("\n")], { type: "text/csv;charset=utf-8" });
   417→   417→      const url = URL.createObjectURL(blob);
   418→   418→      const a = document.createElement("a");
   419→   419→      a.href = url;
   420→   420→      a.download = `SentinelAI-incidents-${new Date().toISOString().slice(0, 10)}.csv`;
   421→   421→      document.body.appendChild(a);
   422→   422→      a.click();
   423→   423→      a.remove();
   424→   424→      URL.revokeObjectURL(url);
   425→   425→      toast.success(`Exported ${list.length} incidents to CSV`);
   426→   426→    } catch (err) {
   427→   427→      toast.error("CSV export failed", {

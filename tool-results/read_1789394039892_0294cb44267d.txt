    70→function priorityOf(severity: string): { label: string; chip: string } {
    71→  switch (severity) {
    72→    case "Critical":
    73→      return { label: "P1", chip: "border-red-500/40 bg-red-500/10 text-red-300" };
    74→    case "High":
    75→      return { label: "P2", chip: "border-amber-500/40 bg-amber-500/10 text-amber-300" };
    76→    case "Medium":
    77→      return { label: "P3", chip: "border-yellow-400/40 bg-yellow-400/10 text-yellow-300" };
    78→    case "Low":
    79→      return { label: "P4", chip: "border-emerald-400/40 bg-emerald-400/10 text-emerald-300" };
    80→    default:
    81→      return { label: "FP", chip: "border-slate-500/40 bg-slate-500/10 text-slate-300" };
    82→  }
    83→}

   108→function KpiCard({
   109→  icon: Icon,
   110→  label,
   111→  value,
   112→  sub,
   113→  tone,
   114→  delay,
   115→}: {
   116→  icon: typeof Inbox;
   117→  label: string;
   118→  value: number;
   119→  sub?: string;
   120→  tone: keyof typeof KPI_TONES;
   121→  delay: number;
   122→}) {
   123→  const t = KPI_TONES[tone];
   124→  return (
   125→    <motion.div {...fadeUp} transition={{ duration: 0.35, delay }}>
   126→      <Card className={cn("border-l-2 p-4 transition-colors hover:border-emerald-500/30 hover:bg-card/80", t.border)}>
   127→        {/* header row: label + icon; value/sub get the full card width below (no truncation) */}
   128→        <div className="flex items-start justify-between gap-2">
   129→          <p className="whitespace-nowrap text-[10px] font-medium uppercase tracking-[0.08em] text-muted-foreground sm:text-[11px]">{label}</p>
   130→          <span className={cn("flex size-9 shrink-0 items-center justify-center rounded-lg border", t.icon)}>
   131→            <Icon className="size-4.5" aria-hidden="true" />
   132→          </span>
   133→        </div>
   134→        <p className={cn("mt-0.5 font-mono text-3xl font-bold tabular-nums", t.value)}>
   135→          {formatCount(value)}
   136→        </p>
   137→        {sub && <p className="mt-0.5 truncate text-[11px] text-muted-foreground" title={sub}>{sub}</p>}
   138→      </Card>
   139→    </motion.div>
   140→  );
   141→}

   172→function SeverityDonut({ data }: { data: DashboardSummary }) {
   173→  const severityOrder = ["Critical", "High", "Medium", "Low", "False Positive"];
   174→  const rows = severityOrder
   175→    .map((sev) => ({
   176→      severity: sev,
   177→      count: data.alertsBySeverity.find((s) => s.severity === sev)?.count ?? 0,
   178→    }))
   179→    .filter((r) => r.count > 0);
   180→  const hasData = rows.length > 0;
   181→
   182→  return (
   183→      <Card className="gap-3 rounded-xl">
   184→      <CardHeader className="pb-2">
   185→        <CardTitle className="text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   186→          Alerts by Severity
   187→        </CardTitle>
   188→      </CardHeader>
   189→      <CardContent>
   190→        {hasData ? (
   191→          <div className="relative h-56">
   192→            <ResponsiveContainer width="100%" height="100%">
   193→              <PieChart>
   194→                <Pie
   195→                  data={rows}
   196→                  dataKey="count"
   197→                  nameKey="severity"
   198→                  innerRadius="62%"
   199→                  outerRadius="88%"
   200→                  paddingAngle={3}
   201→                  strokeWidth={0}
   202→                >
   203→                  {rows.map((row) => (
   204→                    <Cell key={row.severity} fill={severityStyle(row.severity).hex} />
   205→                  ))}
   206→                </Pie>
   207→                <Tooltip contentStyle={TOOLTIP_STYLE} itemStyle={{ color: "var(--popover-foreground)" }} />
   208→              </PieChart>
   209→            </ResponsiveContainer>
   210→            <div className="pointer-events-none absolute inset-0 flex flex-col items-center justify-center">
   211→              <span className="font-mono text-3xl font-bold tabular-nums">{formatCount(data.totalAlerts)}</span>
   212→              <span className="text-[10px] uppercase tracking-[0.2em] text-muted-foreground">alerts</span>
   213→            </div>
   214→          </div>
   215→        ) : (
   216→          <div className="flex h-56 items-center justify-center text-sm text-muted-foreground">
   217→            No alerts yet
   218→          </div>
   219→        )}
   220→        <ul className="mt-3 flex flex-wrap justify-center gap-x-4 gap-y-1.5">
   221→          {severityOrder.map((sev) => {
   222→            const count = data.alertsBySeverity.find((s) => s.severity === sev)?.count ?? 0;
   223→            const s = severityStyle(sev);
   224→            return (
   225→              <li key={sev} className="flex items-center gap-1.5 text-xs text-muted-foreground">
   226→                <span className={cn("size-2 rounded-full", s.dot)} aria-hidden="true" />
   227→                {sev === "False Positive" ? "False Pos." : sev}
   228→                <span className="font-mono tabular-nums text-foreground/80">{count}</span>
   229→              </li>
   230→            );
   231→          })}
   232→        </ul>
   233→      </CardContent>
   234→    </Card>
   235→  );
   236→}

   300→      <CardHeader className="pb-2">
   301→        <CardTitle className="text-sm font-semibold uppercase tracking-wider text-muted-foreground">

   303→        </CardTitle>
   304→        <CardDescription className="text-xs">Sensor coverage across the estate</CardDescription>
   305→      </CardHeader>
   306→      <CardContent>
   307→        {rows.length > 0 ? (
   308→          <ul className="flex flex-col gap-3">
   309→            {rows.slice(0, 7).map((r) => (
   310→              <li key={r.source} className="flex items-center gap-3">
   311→                <span className="w-24 shrink-0 truncate text-xs text-foreground/85" title={r.sourceLabel}>

   356→function IncidentTable({ incidents }: { incidents: IncidentDTO[] }) {
   357→  const openIncident = useSocStore((s) => s.openIncident);
   358→  const [exporting, setExporting] = useState(false);
   359→  const [exportingPdf, setExportingPdf] = useState(false);
   360→
   361→  const fetchAllIncidents = async (): Promise<IncidentDTO[]> => {
   362→    const all = await apiGet<unknown>("/api/incidents");
   363→    return Array.isArray(all)
   364→      ? (all as IncidentDTO[])
   365→      : ((all as { incidents?: IncidentDTO[] }).incidents ?? []);
   366→  };
   367→
   368→  const exportBriefing = async () => {
   369→    setExportingPdf(true);
   370→    try {
   371→      const list = await fetchAllIncidents();
   372→      if (list.length === 0) {
   373→        toast.info("Nothing to brief — no incidents yet.");
   374→        return;
   375→      }
   376→      const summary = await apiGet<DashboardSummary>("/api/dashboard/summary");
   377→      const { downloadBriefingPackPdf } = await import("@/lib/report-pdf");
   378→      await downloadBriefingPackPdf(list, {
   379→        totalAlerts: summary.totalAlerts,
   380→        unacknowledgedAlerts: summary.counts.unacknowledgedAlerts,
   381→      });
   382→      toast.success(`Briefing pack exported — ${list.length} incidents`);
   383→    } catch (err) {
   384→      toast.error("Briefing export failed", {
   385→        description: err instanceof Error ? err.message : undefined,
   386→      });
   387→    } finally {
   388→      setExportingPdf(false);
   389→    }
   390→  };
   391→
   392→  const exportCsv = async () => {
   393→    setExporting(true);
   394→    try {
   395→      const list = await fetchAllIncidents();
   396→      if (list.length === 0) {
   397→        toast.info("Nothing to export — no incidents yet.");
   398→        return;
   399→      }
   400→      const esc = (v: unknown) => {
   401→        const s = String(v ?? "");
   402→        return /["\n,]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
   403→      };
   404→      const header = [
   405→        "incident_id", "title", "severity", "classification", "threat_score",
   406→        "confidence_pct", "status", "alert_count", "sources", "ai_analyzed",
   407→        "created_at", "updated_at",
   408→      ].join(",");
   409→      const rows = list.map((i) =>
   410→        [
   411→          i.incidentId, i.title, i.severity, i.classification, i.threatScore,
   412→          i.confidence, i.status, i.alertCount, i.sources.join("; "), i.analyzed ? "yes" : "no",
   413→          i.createdAt, i.updatedAt,
   414→        ].map(esc).join(",")
   415→      );
   416→      const blob = new Blob([[header, ...rows].join("\n")], { type: "text/csv;charset=utf-8" });
   417→      const url = URL.createObjectURL(blob);
   418→      const a = document.createElement("a");
   419→      a.href = url;
   420→      a.download = `SentinelAI-incidents-${new Date().toISOString().slice(0, 10)}.csv`;
   421→      document.body.appendChild(a);
   422→      a.click();
   423→      a.remove();
   424→      URL.revokeObjectURL(url);
   425→      toast.success(`Exported ${list.length} incidents to CSV`);
   426→    } catch (err) {
   427→      toast.error("CSV export failed", {
   428→        description: err instanceof Error ? err.message : undefined,
   429→      });
   430→    } finally {
   431→      setExporting(false);
   432→    }
   433→  };
   434→
   435→  return (
   436→    <motion.section {...fadeUp} transition={{ duration: 0.4, delay: 0.25 }} aria-label="Priority incidents">
   437→      <Card className="gap-0 rounded-xl p-0">
   438→        <CardHeader className="border-b border-border/70 py-4">
   439→          <div className="flex items-center justify-between gap-2">
   440→            <div>
   441→              <CardTitle className="flex items-center gap-2 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   442→                <Siren className="size-4 text-red-400" aria-hidden="true" />
   443→                Priority Incidents
   444→              </CardTitle>
   445→              <CardDescription className="mt-1 text-xs">
   446→                Ranked by threat score — click a row to open the investigation view
   447→              </CardDescription>
   448→            </div>
   449→            <div className="flex shrink-0 items-center gap-2">
   450→              <Button
   451→                type="button"
   452→                variant="outline"
   453→                size="sm"
   454→                className="min-h-9 gap-1.5 border-border text-xs text-muted-foreground hover:border-emerald-500/40 hover:text-emerald-300"
   455→                onClick={() => void exportBriefing()}
   456→                disabled={exportingPdf}
   457→                aria-label="Export situation briefing pack as PDF"
   458→              >
   459→                <FileText className={exportingPdf ? "size-3.5 animate-pulse" : "size-3.5"} aria-hidden="true" />
   460→                Briefing .pdf
   461→              </Button>
   462→              <Button
   463→                type="button"
   464→                variant="outline"
   465→                size="sm"
   466→                className="min-h-9 gap-1.5 border-border text-xs text-muted-foreground hover:border-emerald-500/40 hover:text-emerald-300"
   467→                onClick={() => void exportCsv()}
   468→                disabled={exporting}
   469→                aria-label="Export all incidents as CSV"
   470→              >
   471→                <FileDown className={exporting ? "size-3.5 animate-pulse" : "size-3.5"} aria-hidden="true" />
   472→                Export CSV
   473→              </Button>
   474→            </div>
   475→          </div>
   476→        </CardHeader>
   477→        <CardContent className="p-0">
   478→          {incidents.length === 0 ? (
   479→            <p className="py-10 text-center text-sm text-muted-foreground">
   480→              No incidents correlated yet — alerts will be grouped automatically.
   481→            </p>
   482→          ) : (
   483→            <div className="soc-scroll max-h-[520px] overflow-auto">
   484→              <Table>
   485→                <TableHeader className="sticky top-0 z-10 bg-card/95 backdrop-blur">
   486→                  <TableRow className="hover:bg-transparent">
   487→                    <TableHead className="w-14 pl-4 text-[11px] uppercase tracking-wider">Prio</TableHead>
   488→                    <TableHead className="text-[11px] uppercase tracking-wider">Incident</TableHead>
   489→                    <TableHead className="min-w-44 text-[11px] uppercase tracking-wider">Title</TableHead>
   490→                    <TableHead className="text-[11px] uppercase tracking-wider">Severity</TableHead>
   491→                    <TableHead className="min-w-32 text-[11px] uppercase tracking-wider">Threat Score</TableHead>
   492→                    <TableHead className="text-right text-[11px] uppercase tracking-wider">Conf.</TableHead>
   493→                    <TableHead className="text-right text-[11px] uppercase tracking-wider">Alerts</TableHead>
   494→                    <TableHead className="pr-4 text-[11px] uppercase tracking-wider">Status</TableHead>
   495→                    <TableHead className="w-10 pr-3 text-[11px] uppercase tracking-wider">AI</TableHead>
   496→                  </TableRow>
   497→                </TableHeader>
   498→                <TableBody>
   499→                  {incidents.map((inc) => {
   500→                    const prio = priorityOf(inc.severity);
   501→                    return (
   502→                      <TableRow

   657→export function CommandCenter() {
   658→  const query = useQuery({

   660→    queryFn: () => apiGet<DashboardSummary>("/api/dashboard/summary"),

   667→      <ErrorState
   668→        message={query.error instanceof Error ? query.error.message : undefined}
   669→        onRetry={() => query.refetch()}
   670→      />
   671→    );
   672→  }
   673→  if (!query.data || query.data.totalAlerts === 0) {
   674→    return <EmptyHero />;
   675→  }
   676→
   677→  return (
   678→    <div className="flex flex-col gap-4 xl:flex-row xl:items-start xl:gap-6">
   679→      <div className="flex min-w-0 flex-1 flex-col gap-4 sm:gap-6">

   684→      </div>

   686→    </div>

   314→/** Build the full plain-text incident report for download/export. */
   315→function buildReportText(incident: IncidentDetailDTO): string {
   316→  const rule = "-".repeat(72);
   317→  const lines: string[] = [];
   318→  lines.push("=".repeat(72));
   319→  lines.push(`SENTINELAI INCIDENT REPORT — ${incident.incidentId}`);
   320→  lines.push("=".repeat(72));
   321→  lines.push(`Title          : ${incident.title}`);
   322→  lines.push(`Severity       : ${incident.severity}`);
   323→  lines.push(`Classification : ${incident.classification}`);
   324→  lines.push(`Threat Score   : ${incident.threatScore}/100`);
   325→  lines.push(`Confidence     : ${incident.confidence}%`);
   326→  lines.push(`Status         : ${incident.status}`);
   327→  lines.push(`Related Alerts : ${incident.alertCount} (${incident.sources.join(", ")})`);
   328→  lines.push(`Generated      : ${new Date().toISOString()}`);
   329→  lines.push("");
   330→  lines.push(rule);
   331→  lines.push("MITRE ATT&CK MAPPING");
   332→  lines.push(rule);
   333→  if (incident.mitre.length === 0) lines.push("(none mapped)");
   334→  for (const t of incident.mitre) lines.push(`  ${t.id}  ${t.name} [${t.tactic}]`);
   335→  lines.push("");
   336→  lines.push(rule);
   337→  lines.push("RISK SIGNALS");
   338→  lines.push(rule);
   339→  if (incident.riskSignals.length === 0) lines.push("(none)");
   340→  for (const r of incident.riskSignals) lines.push(`  +${r.points}  ${r.signal}`);
   341→  lines.push("");
   342→  lines.push(rule);
   343→  lines.push("BLUF — BOTTOM LINE UP FRONT");
   344→  lines.push(rule);
   345→  lines.push(incident.bluf ?? "");
   346→  if (incident.explanation) {
   347→    lines.push("");
   348→    lines.push(rule);
   349→    lines.push("ANALYST EXPLANATION");
   350→    lines.push(rule);
   351→    lines.push(incident.explanation);
   352→  }
   353→  if (incident.recommendedActions.length > 0) {
   354→    lines.push("");
   355→    lines.push(rule);
   356→    lines.push("RECOMMENDED ACTIONS");
   357→    lines.push(rule);
   358→    incident.recommendedActions.forEach((a, i) => lines.push(`  ${i + 1}. ${a}`));
   359→  }
   360→  if (incident.evidence.length > 0) {
   361→    lines.push("");
   362→    lines.push(rule);
   363→    lines.push("KEY EVIDENCE");
   364→    lines.push(rule);
   365→    for (const e of incident.evidence) lines.push(`  - ${e}`);
   366→  }
   367→  lines.push("");
   368→  lines.push(rule);
   369→  lines.push("RELATED ALERT TIMELINE");
   370→  lines.push(rule);
   371→  for (const a of incident.alerts) {
   372→    lines.push(`  ${a.timestamp}  [${a.sourceLabel}]  ${a.event}  (${a.alertId})`);
   373→    lines.push(`    ${a.description}`);
   374→  }
   375→  lines.push("");
   376→  lines.push("SentinelAI — simulated demo data, not for operational use.");
   377→  return lines.join("\n");
   378→}

   380→function BlufReport({ incident }: { incident: IncidentDetailDTO }) {
   381→  const [copied, setCopied] = useState(false);
   382→  if (!incident.bluf) return null;
   383→
   384→  const copyBluf = async () => {
   385→    try {
   386→      await navigator.clipboard.writeText(incident.bluf ?? "");
   387→      setCopied(true);
   388→      toast.success("BLUF summary copied to clipboard");
   389→      setTimeout(() => setCopied(false), 2000);
   390→    } catch {
   391→      toast.error("Clipboard unavailable in this browser");
   392→    }
   393→  };
   394→
   395→  const downloadReport = () => {
   396→    const blob = new Blob([buildReportText(incident)], { type: "text/plain;charset=utf-8" });
   397→    const url = URL.createObjectURL(blob);
   398→    const a = document.createElement("a");
   399→    a.href = url;
   400→    a.download = `SentinelAI-report-${incident.incidentId}.txt`;
   401→    document.body.appendChild(a);
   402→    a.click();
   403→    a.remove();
   404→    URL.revokeObjectURL(url);
   405→    toast.success(`Full report downloaded (${incident.incidentId})`);
   406→  };
   407→
   408→  return (
   409→    <motion.div
   410→      initial={{ opacity: 0, y: 10 }}
   411→      animate={{ opacity: 1, y: 0 }}
   412→      transition={{ duration: 0.4 }}
   413→    >
   414→      <Card className="overflow-hidden rounded-xl border-emerald-500/25 p-0">
   415→        <div className="flex flex-wrap items-center gap-2 border-b border-emerald-500/25 bg-emerald-500/10 px-4 py-2.5">
   416→          <FileText className="size-4 text-emerald-400" aria-hidden="true" />

   502→function DetailPane({ incidentId }: { incidentId: string }) {
   503→  const queryClient = useQueryClient();
   504→
   505→  const detailQuery = useQuery({
   506→    queryKey: ["incident", incidentId],
   507→    queryFn: async () => {
   508→      const d = await apiGet<{ incident: IncidentDetailDTO }>(`/api/incidents/${incidentId}`);
   509→      return d.incident;
   510→    },
   511→    enabled: Boolean(incidentId),
   512→  });
   513→
   514→  const [analyzeLine, setAnalyzeLine] = useState(0);
   515→  const analyzeMutation = useMutation({
   516→    mutationFn: () => apiSend(`/api/incidents/${incidentId}/analyze`, "POST"),
   517→    onSuccess: (data: unknown) => {
   518→      let msg = "AI analysis complete — BLUF report ready.";
   519→      if (data && typeof data === "object" && "message" in data) {
   520→        const m = (data as { message: unknown }).message;
   521→        if (typeof m === "string") msg = m;
   522→      }
   523→      toast.success("Analysis complete", { description: msg });
   524→      queryClient.invalidateQueries({ queryKey: ["incident", incidentId] });
   525→      queryClient.invalidateQueries({ queryKey: ["incidents"] });
   526→      queryClient.invalidateQueries({ queryKey: ["dashboard-summary"] });
   527→    },
   528→    onError: (err: Error) => {
   529→      toast.error("AI analysis failed", { description: err.message });
   530→    },
   531→  });
   532→
   533→  useEffect(() => {
   534→    if (!analyzeMutation.isPending) return;
   535→    const t = setInterval(
   536→      () => setAnalyzeLine((i) => (i + 1) % ANALYZE_LINES.length),
   537→      2500
   538→    );
   539→    return () => clearInterval(t);
   540→  }, [analyzeMutation.isPending]);
   541→
   542→  const updateMutation = useMutation({
   543→    mutationFn: (payload: IncidentUpdatePayload) =>
   544→      apiSend<unknown>(`/api/incidents/${incidentId}`, "PATCH", payload),
   545→    onSuccess: (_data, variables) => {
   546→      const what = variables.status
   547→        ? `Status set to ${variables.status}`
   548→        : variables.classification
   549→          ? `Classification set to ${variables.classification}`
   550→          : "Incident updated";
   551→      toast.success(what, { description: "Analyst feedback recorded." });
   552→      queryClient.invalidateQueries({ queryKey: ["incident", incidentId] });
   553→      queryClient.invalidateQueries({ queryKey: ["incidents"] });
   554→      queryClient.invalidateQueries({ queryKey: ["dashboard-summary"] });
   555→    },
   556→    onError: (err: Error) => {
   557→      toast.error("Update failed", { description: err.message });
   558→    },
   559→  });
   560→
   561→  if (detailQuery.isLoading) {
   562→    return (
   563→      <div className="flex flex-col gap-4 sm:gap-6" aria-busy="true" aria-label="Loading incident detail">
   564→        <Skeleton className="h-36 rounded-xl" />
   565→        <Skeleton className="h-52 rounded-xl" />
   566→        <Skeleton className="h-64 rounded-xl" />
   567→      </div>
   568→    );
   569→  }
   570→  if (detailQuery.isError) {
   571→    return (
   572→      <ErrorState
   573→        message={detailQuery.error instanceof Error ? detailQuery.error.message : undefined}
   574→        onRetry={() => detailQuery.refetch()}
   575→      />
   576→    );
   577→  }
   578→  const incident = detailQuery.data;
   579→  if (!incident) return null;
   580→
   581→  return (
   582→    <div className="flex min-w-0 flex-col gap-4 sm:gap-6">
   583→      {/* Header */}
   584→      <motion.div
   585→        key={incident.id}
   586→        initial={{ opacity: 0, y: 10 }}
   587→        animate={{ opacity: 1, y: 0 }}
   588→        transition={{ duration: 0.35 }}
   589→      >
   590→        <Card className="rounded-xl p-4 sm:p-6">
   591→          <div className="flex flex-col gap-4">
   592→            <div className="flex flex-wrap items-center gap-2">
   593→              <SeverityBadge severity={incident.severity} />
   594→              <ClassificationBadge classification={incident.classification} />
   595→              <StatusBadge status={incident.status} />
   596→              <span className="ml-auto font-mono text-sm font-bold text-emerald-300/90">
   597→                {incident.incidentId}
   598→              </span>
   599→            </div>
   600→            <div>
   601→              <h2 className="text-lg font-bold leading-snug tracking-tight sm:text-xl">
   602→                {incident.title}
   603→              </h2>
   604→              <p className="mt-1 text-xs text-muted-foreground">
   605→                {incident.alertCount} related alert{incident.alertCount === 1 ? "" : "s"}
   606→                {" · "}
   607→                {incident.sources.join(", ") || "single source"}
   608→                {" · "}
   609→                created {timeAgo(incident.createdAt)}
   610→              </p>
   611→            </div>
   612→
   613→            {/* Actions */}
   614→            <div className="flex flex-wrap items-center gap-2 border-t border-border/70 pt-4">
   615→              <Button
   616→                type="button"
   617→                onClick={() => {
   618→                  setAnalyzeLine(0);
   619→                  analyzeMutation.mutate();
   620→                }}
   621→                disabled={analyzeMutation.isPending}
   622→                className="min-h-11 gap-2 border-emerald-500/50 bg-emerald-500/15 font-semibold text-emerald-300 hover:bg-emerald-500/25 hover:text-emerald-200"
   623→                aria-label="Run AI analysis on this incident"
   624→              >
   625→                {analyzeMutation.isPending ? (
   626→                  <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   627→                ) : (
   628→                  <Sparkles className="size-4" aria-hidden="true" />
   629→                )}
   630→                {analyzeMutation.isPending ? (
   631→                  <span className="font-mono text-xs">{ANALYZE_LINES[analyzeLine]}</span>
   632→                ) : (
   633→                  "Run AI Analysis"
   634→                )}
   635→              </Button>
   636→
   637→              <div className="flex items-center gap-2">
   638→                <label htmlFor="workflow-status" className="sr-only">Incident status</label>
   639→                <Select
   640→                  value={incident.status}
   641→                  onValueChange={(v) =>
   642→                    updateMutation.mutate({ status: v as IncidentStatus })
   643→                  }
   644→                  disabled={updateMutation.isPending}
   645→                >
   646→                  <SelectTrigger id="workflow-status" className="min-h-11 w-40" aria-label="Set incident status">
   647→                    <SelectValue placeholder="Status" />
   648→                  </SelectTrigger>
   649→                  <SelectContent>
   650→                    {STATUSES.map((s) => (
   651→                      <SelectItem key={s} value={s}>
   652→                        {s}
   653→                      </SelectItem>
   654→                    ))}
   655→                  </SelectContent>
   656→                </Select>
   657→              </div>
   658→
   659→              <div className="ml-auto flex items-center gap-1.5">
   660→                <Button
   661→                  type="button"
   662→                  variant="outline"
   663→                  size="sm"
   664→                  className="min-h-11 gap-1.5 border-emerald-500/40 text-emerald-300 hover:bg-emerald-500/10 hover:text-emerald-200"
   665→                  disabled={updateMutation.isPending}
   666→                  onClick={() =>
   667→                    updateMutation.mutate({ classification: "Genuine Threat" as Classification })
   668→                  }
   669→                  aria-label="Confirm this incident as a genuine threat"
   670→                >
   671→                  <ThumbsUp className="size-3.5" aria-hidden="true" />
   672→                  Confirm Threat
   673→                </Button>
   674→                <Button
   675→                  type="button"
   676→                  variant="outline"
   677→                  size="sm"
   678→                  className="min-h-11 gap-1.5 border-slate-500/40 text-slate-300 hover:bg-slate-500/10 hover:text-slate-200"
   679→                  disabled={updateMutation.isPending}
   680→                  onClick={() =>
   681→                    updateMutation.mutate({ classification: "False Positive" as Classification })
   682→                  }
   683→                  aria-label="Mark this incident as a false positive"
   684→                >
   685→                  <ThumbsDown className="size-3.5" aria-hidden="true" />
   686→                  Mark False Positive
   687→                </Button>
   688→                <Button
   689→                  type="button"
   690→                  variant="ghost"
   691→                  size="icon"
   692→                  className="size-11 shrink-0 text-muted-foreground"
   693→                  aria-label="Dismiss feedback (no-op in demo)"
   694→                  title="Dismiss — no-op in this demo"
   695→                >
   696→                  <X className="size-4" aria-hidden="true" />
   697→                </Button>
   698→              </div>
   699→            </div>
   700→          </div>
   701→        </Card>
   702→      </motion.div>
   703→
   704→      {/* Unanalyzed ribbon */}
   705→      {!incident.analyzed && (
   706→        <div
   707→          role="status"
   708→          className="flex items-center gap-2 rounded-xl border border-amber-500/30 bg-amber-500/10 px-4 py-3 text-xs text-amber-300"
   709→        >
   710→          <Sparkles className="size-4 shrink-0" aria-hidden="true" />
   711→          Deterministic scoring only — run AI Analysis for the full LLM investigation report.
   712→        </div>
   713→      )}
   714→
   715→      <ScorePanel incident={incident} />
   716→
   717→      <div className="grid grid-cols-1 gap-4 sm:gap-6 xl:grid-cols-2">
   718→        <AttackTimeline incident={incident} />
   719→        <div className="flex flex-col gap-4 sm:gap-6">
   720→          <MitrePanel incident={incident} />
   721→          {incident.explanation && (
   722→            <Card className="rounded-xl">
   723→              <CardHeader className="pb-3">
   724→                <CardTitle className="flex items-center gap-2 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   725→                  <ShieldCheck className="size-4 text-emerald-400" aria-hidden="true" />
   726→                  Analyst Explanation
   727→                </CardTitle>
   728→              </CardHeader>
   729→              <CardContent>
   730→                <p className="whitespace-pre-wrap text-sm leading-relaxed text-foreground/85">
   731→                  {incident.explanation}
   732→                </p>
   733→              </CardContent>
   734→            </Card>
   735→          )}
   736→        </div>
   737→      </div>
   738→
   739→      <BlufReport incident={incident} />
   740→    </div>
   741→  );
   742→}

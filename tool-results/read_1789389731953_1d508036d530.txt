    88→interface FeedFilters {
    89→  search: string;
    90→  source: string;
    91→  severity: string;

    93→  hideAck: boolean;

    95→
    96→const DEFAULT_FILTERS: FeedFilters = {
    97→  search: "",
    98→  source: "all",
    99→  severity: "all",
   100→  correlated: "all",
   101→  hideAck: false,
   102→};
   103→
   104→function buildAlertsUrl(filters: FeedFilters): string {
   105→  const params = new URLSearchParams();
   106→  if (filters.search.trim()) params.set("search", filters.search.trim());
   107→  if (filters.source !== "all") params.set("source", filters.source);
   108→  if (filters.severity !== "all") params.set("severity", filters.severity);
   109→  if (filters.correlated === "correlated") params.set("correlated", "true");
   110→  if (filters.hideAck) params.set("ack", "unack");
   111→  const qs = params.toString();
   112→  return qs ? `/api/alerts?${qs}` : "/api/alerts";
   113→}

   119→interface FeedPreset {

   125→const PRESETS_KEY = "sentinelai.feedPresets";
   126→
   127→function loadPresets(): FeedPreset[] {
   128→  if (typeof window === "undefined") return []; // SSR guard
   129→  try {
   130→    const raw = localStorage.getItem(PRESETS_KEY);
   131→    if (!raw) return [];
   132→    const parsed = JSON.parse(raw) as unknown;
   133→    if (!Array.isArray(parsed)) return [];
   134→    return parsed.filter(
   135→      (p): p is FeedPreset =>
   136→        typeof p === "object" && p !== null &&
   137→        typeof (p as FeedPreset).id === "string" &&
   138→        typeof (p as FeedPreset).name === "string" &&
   139→        typeof (p as FeedPreset).filters === "object"
   140→    );
   141→  } catch {
   142→    return [];
   143→  }
   144→}
   145→
   146→function savePresets(presets: FeedPreset[]) {
   147→  try {
   148→    localStorage.setItem(PRESETS_KEY, JSON.stringify(presets));
   149→  } catch {
   150→    // storage full/blocked — presets just won't persist
   151→  }
   152→}
   153→
   154→function filtersMatch(a: FeedFilters, b: FeedFilters): boolean {
   155→  return (
   156→    a.search === b.search &&
   157→    a.source === b.source &&
   158→    a.severity === b.severity &&
   159→    a.correlated === b.correlated &&
   160→    a.hideAck === b.hideAck
   161→  );
   162→}

   170→function guessFormatFromName(name: string): NonNullable<ImportPayload["format"]> | null {
   171→  const ext = name.toLowerCase().split(".").pop() ?? "";
   172→  if (ext === "json") return "json";
   173→  if (ext === "csv" || ext === "tsv") return "csv";
   174→  if (ext === "txt" || ext === "log" || ext === "text") return "text";
   175→  return null;
   176→}

   371→/** Alerts newer than this get a pulsing "NEW" chip. */
   372→const FRESH_MS = 120_000;
   373→
   374→function AlertRow({
   375→  alert,
   376→  incidentIdMap,
   377→  ackPending,
   378→  onToggleAck,
   379→}: {
   380→  alert: AlertDTO;
   381→  incidentIdMap: Map<string, string>;
   382→  ackPending: boolean;
   383→  onToggleAck: (alert: AlertDTO) => void;
   384→}) {
   385→  const openIncident = useSocStore((s) => s.openIncident);
   386→  const displayIncidentId = alert.incidentId
   387→    ? incidentIdMap.get(alert.incidentId) ?? "View incident"
   388→    : null;
   389→  const fresh = Date.now() - new Date(alert.timestamp).getTime() < FRESH_MS;
   390→
   391→  return (
   392→    <TableRow
   393→      className={cn(
   394→        "border-white/5 transition-opacity hover:bg-emerald-500/5",
   395→        alert.acknowledged && "bg-muted/20 opacity-55 hover:opacity-80"
   396→      )}
   397→    >
   398→      <TableCell className="whitespace-nowrap font-mono text-xs text-muted-foreground" title={formatDateTime(alert.timestamp)}>
   399→        <span className="block text-foreground/80">{timeAgo(alert.timestamp)}</span>
   400→        <span className="block text-[10px] text-muted-foreground/70">{formatDateTime(alert.timestamp)}</span>
   401→      </TableCell>
   402→      <TableCell className="whitespace-nowrap font-mono text-xs font-semibold text-foreground/85">
   403→        <span className="flex items-center gap-1.5">
   404→          {alert.alertId}
   405→          {fresh && !alert.acknowledged && (
   406→            <span className="inline-flex items-center gap-1 rounded-full border border-emerald-500/50 bg-emerald-500/15 px-1.5 py-px font-mono text-[9px] font-bold tracking-wider text-emerald-300">
   407→              <span className="relative flex size-1.5">
   408→                <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-400 opacity-70" aria-hidden="true" />
   409→                <span className="relative inline-flex size-1.5 rounded-full bg-emerald-400" aria-hidden="true" />
   410→              </span>
   411→              NEW
   412→            </span>
   413→          )}
   414→          {alert.acknowledged && (
   415→            <span className="rounded border border-emerald-500/40 bg-emerald-500/10 px-1 py-px font-mono text-[9px] font-bold tracking-wider text-emerald-400/90">
   416→              ACK
   417→            </span>
   418→          )}
   419→        </span>
   420→      </TableCell>
   421→      <TableCell>
   422→        <SourceChip source={alert.source} sourceLabel={alert.sourceLabel} />
   423→      </TableCell>
   424→      <TableCell className="font-mono text-[11px] leading-4 text-muted-foreground">
   425→        {alert.user && <span className="block truncate" title={`user: ${alert.user}`}>u: {alert.user}</span>}
   426→        {alert.device && <span className="block truncate" title={`device: ${alert.device}`}>d: {alert.device}</span>}
   427→        {alert.ip && <span className="block truncate" title={`ip: ${alert.ip}`}>ip: {alert.ip}</span>}
   428→        {!alert.user && !alert.device && !alert.ip && <span className="text-muted-foreground/50">—</span>}
   429→      </TableCell>
   430→      <TableCell>
   431→        <EventChip event={alert.event} />
   432→      </TableCell>
   433→      <TableCell className="max-w-64">
   434→        <span className="block truncate text-xs text-foreground/85" title={alert.description}>
   435→          {alert.description}
   436→        </span>
   437→      </TableCell>
   438→      <TableCell>
   439→        <RawSeverityBadge raw={alert.rawSeverity} />
   440→      </TableCell>
   441→      <TableCell>
   442→        <CorrelationChip
   443→          incidentId={displayIncidentId}
   444→          onClick={
   445→            alert.incidentId ? () => openIncident(alert.incidentId as string) : undefined
   446→          }
   447→        />
   448→      </TableCell>
   449→      <TableCell className="pr-3">
   450→        <button
   451→          type="button"
   452→          onClick={() => onToggleAck(alert)}
   453→          disabled={ackPending}
   454→          aria-pressed={alert.acknowledged}
   455→          aria-label={
   456→            alert.acknowledged
   457→              ? `Clear acknowledgement for alert ${alert.alertId}`
   458→              : `Acknowledge alert ${alert.alertId}`
   459→          }
   460→          title={
   461→            alert.acknowledged
   462→              ? `Acknowledged — click to clear`
   463→              : `Mark ${alert.alertId} as reviewed`
   464→          }
   465→          className={cn(
   466→            "inline-flex size-8 items-center justify-center rounded-lg border transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring",
   467→            alert.acknowledged
   468→              ? "border-emerald-500/50 bg-emerald-500/15 text-emerald-300 hover:bg-emerald-500/25"
   469→              : "border-border bg-transparent text-muted-foreground hover:border-emerald-500/40 hover:bg-emerald-500/10 hover:text-emerald-300"
   470→          )}
   471→        >
   472→          {ackPending ? (
   473→            <RotateCw className="size-3.5 animate-spin" aria-hidden="true" />
   474→          ) : alert.acknowledged ? (
   475→            <CheckCheck className="size-4" aria-hidden="true" />
   476→          ) : (
   477→            <Check className="size-4" aria-hidden="true" />
   478→          )}
   479→        </button>
   480→      </TableCell>
   481→    </TableRow>
   482→  );
   483→}

   499→export function ThreatFeed() {
   500→  const [filters, setFilters] = useState<FeedFilters>(DEFAULT_FILTERS);
   501→  const [searchInput, setSearchInput] = useState("");
   502→  const [showImport, setShowImport] = useState(false);
   503→  const [presets, setPresets] = useState<FeedPreset[]>(() => loadPresets());
   504→  const [presetDialogOpen, setPresetDialogOpen] = useState(false);
   505→  const [presetName, setPresetName] = useState("");
   506→  const queryClient = useQueryClient();
   507→
   508→  // track whether current filters match a saved preset (for the active chip)
   509→  const activePreset = useMemo(
   510→    () => presets.find((p) => filtersMatch(p.filters, filters)) ?? null,
   511→    [presets, filters]
   512→  );
   513→
   514→  const applyPreset = (preset: FeedPreset) => {
   515→    setFilters({ ...preset.filters });
   516→    setSearchInput(preset.filters.search);
   517→    toast.success(`Preset applied — ${preset.name}`);
   518→  };
   519→
   520→  const savePreset = () => {
   521→    const name = presetName.trim();
   522→    if (!name) return;
   523→    // overwrite an existing preset with the same name
   524→    const next: FeedPreset[] = [
   525→      ...presets.filter((p) => p.name.toLowerCase() !== name.toLowerCase()),
   526→      { id: `p-${Date.now().toString(36)}`, name, filters: { ...filters } },
   527→    ];
   528→    setPresets(next);
   529→    savePresets(next);
   530→    setPresetDialogOpen(false);
   531→    setPresetName("");
   532→    toast.success(`Preset "${name}" saved`, {
   533→      description: "Stored locally in this browser.",
   534→    });
   535→  };
   536→
   537→  const deletePreset = (preset: FeedPreset) => {
   538→    const next = presets.filter((p) => p.id !== preset.id);
   539→    setPresets(next);
   540→    savePresets(next);
   541→    toast.success(`Preset "${preset.name}" deleted`);
   542→  };
   543→
   544→  // debounce search input into the applied filter
   545→  useEffect(() => {
   546→    const t = setTimeout(() => setFilters((f) => ({ ...f, search: searchInput })), 300);
   547→    return () => clearTimeout(t);
   548→  }, [searchInput]);
   549→
   550→  const alertsQuery = useQuery({
   551→    queryKey: ["alerts", filters],
   552→    queryFn: () => apiGet<unknown>(buildAlertsUrl(filters)),

   581→    queryFn: () => apiGet<unknown>("/api/incidents"),

   584→
   585→  const incidentIdMap = useMemo(() => {

   592→
   593→  return (
   594→    <div className="flex flex-col gap-4 sm:gap-6">

   602→          <div className="flex flex-col gap-3 lg:flex-row lg:items-center">

   633→                </SelectContent>
   634→              </Select>
   635→
   636→              <Label htmlFor="filter-severity" className="sr-only">Filter by severity</Label>

   638→                value={filters.severity}

   641→                <SelectTrigger id="filter-severity" className="min-h-11 w-full sm:w-36" aria-label="Filter by severity">

   647→                  <SelectItem value="high">High</SelectItem>
   648→                  <SelectItem value="medium">Medium</SelectItem>
   649→                  <SelectItem value="low">Low</SelectItem>
   650→                  <SelectItem value="info">Info</SelectItem>
   651→                </SelectContent>
   652→              </Select>
   653→
   654→              <div className="flex min-h-11 items-center gap-2 rounded-lg border border-input bg-background/40 px-3">
   655→                <Switch
   656→                  id="correlated-only"
   657→                  checked={filters.correlated === "correlated"}

   659→                    setFilters((f) => ({
   660→                      ...f,
   661→                      correlated: checked ? "correlated" : "all",

   664→                  aria-label="Show correlated alerts only"

   791→            <button
   792→              type="button"
   793→              onClick={() => setShowImport((v) => !v)}
   794→              className="inline-flex min-h-9 items-center gap-1.5 rounded-md px-2 font-medium text-emerald-400 transition-colors hover:text-emerald-300 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   795→              aria-expanded={showImport}

   838→              <p className="max-w-sm text-xs text-muted-foreground">
   839→                Try clearing the search box, widening the severity filter, or load the
   840→                demo dataset to populate the feed.
   841→              </p>
   842→              <Button
   843→                variant="outline"
   844→                size="sm"
   845→                className="mt-1 min-h-9"
   846→                onClick={() => {
   847→                  setSearchInput("");
   848→                  setFilters(DEFAULT_FILTERS);
   849→                }}
   850→              >
   851→                Clear filters
   852→              </Button>
   853→            </div>
   854→          ) : (
   855→            <div className="soc-scroll max-h-[620px] overflow-auto">
   856→              <Table>
   857→                <TableHeader className="sticky top-0 z-10 bg-card/95 backdrop-blur">
   858→                  <TableRow className="hover:bg-transparent">
   859→                    <TableHead className="pl-4 text-[11px] uppercase tracking-wider">Time</TableHead>
   860→                    <TableHead className="text-[11px] uppercase tracking-wider">Alert ID</TableHead>
   861→                    <TableHead className="text-[11px] uppercase tracking-wider">Source</TableHead>
   862→                    <TableHead className="text-[11px] uppercase tracking-wider">Entity</TableHead>
   863→                    <TableHead className="text-[11px] uppercase tracking-wider">Event</TableHead>
   864→                    <TableHead className="text-[11px] uppercase tracking-wider">Description</TableHead>
   865→                    <TableHead className="text-[11px] uppercase tracking-wider">Raw Sev.</TableHead>

   890→          <DialogHeader>
   891→            <DialogTitle className="flex items-center gap-2">
   892→              <BookmarkPlus className="size-4 text-emerald-400" aria-hidden="true" />
   893→              Save filter preset

   895→            <DialogDescription>
   896→              Stores the current search, source, severity, correlation and triage filters in this

   899→          </DialogHeader>
   900→          <div className="grid gap-2">
   901→            <Label htmlFor="preset-name">Preset name</Label>
   902→            <Input
   903→              id="preset-name"
   904→              value={presetName}
   905→              onChange={(e) => setPresetName(e.target.value)}
   906→              onKeyDown={(e) => {
   907→                if (e.key === "Enter") {
   908→                  e.preventDefault();
   909→                  savePreset();

   912→              placeholder="e.g. Critical auth alerts"
   913→              maxLength={40}
   914→              autoFocus
   915→            />
   916→            <p className="font-mono text-[11px] text-muted-foreground">

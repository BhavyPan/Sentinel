   315→/** Build the full plain-text incident report for download/export. */
   316→function buildReportText(incident: IncidentDetailDTO): string {
   317→  const rule = "-".repeat(72);
   318→  const lines: string[] = [];
   319→  lines.push("=".repeat(72));
   320→  lines.push(`SENTINELAI INCIDENT REPORT — ${incident.incidentId}`);
   321→  lines.push("=".repeat(72));
   322→  lines.push(`Title          : ${incident.title}`);
   323→  lines.push(`Severity       : ${incident.severity}`);
   324→  lines.push(`Classification : ${incident.classification}`);
   325→  lines.push(`Threat Score   : ${incident.threatScore}/100`);
   326→  lines.push(`Confidence     : ${incident.confidence}%`);
   327→  lines.push(`Status         : ${incident.status}`);
   328→  lines.push(`Related Alerts : ${incident.alertCount} (${incident.sources.join(", ")})`);
   329→  lines.push(`Generated      : ${new Date().toISOString()}`);
   330→  lines.push("");
   331→  lines.push(rule);
   332→  lines.push("MITRE ATT&CK MAPPING");
   333→  lines.push(rule);
   334→  if (incident.mitre.length === 0) lines.push("(none mapped)");
   335→  for (const t of incident.mitre) lines.push(`  ${t.id}  ${t.name} [${t.tactic}]`);
   336→  lines.push("");
   337→  lines.push(rule);
   338→  lines.push("RISK SIGNALS");
   339→  lines.push(rule);
   340→  if (incident.riskSignals.length === 0) lines.push("(none)");
   341→  for (const r of incident.riskSignals) lines.push(`  +${r.points}  ${r.signal}`);
   342→  lines.push("");
   343→  lines.push(rule);
   344→  lines.push("BLUF — BOTTOM LINE UP FRONT");
   345→  lines.push(rule);
   346→  lines.push(incident.bluf ?? "");
   347→  if (incident.explanation) {
   348→    lines.push("");
   349→    lines.push(rule);
   350→    lines.push("ANALYST EXPLANATION");
   351→    lines.push(rule);
   352→    lines.push(incident.explanation);
   353→  }
   354→  if (incident.recommendedActions.length > 0) {
   355→    lines.push("");
   356→    lines.push(rule);
   357→    lines.push("RECOMMENDED ACTIONS");
   358→    lines.push(rule);
   359→    incident.recommendedActions.forEach((a, i) => lines.push(`  ${i + 1}. ${a}`));
   360→  }
   361→  if (incident.evidence.length > 0) {
   362→    lines.push("");
   363→    lines.push(rule);
   364→    lines.push("KEY EVIDENCE");
   365→    lines.push(rule);
   366→    for (const e of incident.evidence) lines.push(`  - ${e}`);
   367→  }
   368→  lines.push("");
   369→  lines.push(rule);
   370→  lines.push("RELATED ALERT TIMELINE");
   371→  lines.push(rule);
   372→  for (const a of incident.alerts) {
   373→    lines.push(`  ${a.timestamp}  [${a.sourceLabel}]  ${a.event}  (${a.alertId})`);
   374→    lines.push(`    ${a.description}`);
   375→  }
   376→  lines.push("");
   377→  lines.push("SentinelAI — simulated demo data, not for operational use.");
   378→  return lines.join("\n");
   379→}

   381→function BlufReport({ incident }: { incident: IncidentDetailDTO }) {
   382→  const [copied, setCopied] = useState(false);
   383→  if (!incident.bluf) return null;
   384→
   385→  const copyBluf = async () => {
   386→    try {
   387→      await navigator.clipboard.writeText(incident.bluf ?? "");
   388→      setCopied(true);
   389→      toast.success("BLUF summary copied to clipboard");
   390→      setTimeout(() => setCopied(false), 2000);
   391→    } catch {
   392→      toast.error("Clipboard unavailable in this browser");
   393→    }
   394→  };
   395→
   396→  const downloadReport = () => {
   397→    const blob = new Blob([buildReportText(incident)], { type: "text/plain;charset=utf-8" });
   398→    const url = URL.createObjectURL(blob);
   399→    const a = document.createElement("a");
   400→    a.href = url;
   401→    a.download = `SentinelAI-report-${incident.incidentId}.txt`;
   402→    document.body.appendChild(a);
   403→    a.click();
   404→    a.remove();
   405→    URL.revokeObjectURL(url);
   406→    toast.success(`Full report downloaded (${incident.incidentId})`);
   407→  };
   408→
   409→  return (
   410→    <motion.div
   411→      initial={{ opacity: 0, y: 10 }}
   412→      animate={{ opacity: 1, y: 0 }}
   413→      transition={{ duration: 0.4 }}
   414→    >
   415→      <Card className="overflow-hidden rounded-xl border-emerald-500/25 p-0">

   503→function DetailPane({ incidentId }: { incidentId: string }) {
   504→  const queryClient = useQueryClient();
   505→
   506→  const detailQuery = useQuery({
   507→    queryKey: ["incident", incidentId],
   508→    queryFn: async () => {
   509→      const d = await apiGet<{ incident: IncidentDetailDTO }>(`/api/incidents/${incidentId}`);
   510→      return d.incident;
   511→    },
   512→    enabled: Boolean(incidentId),
   513→  });
   514→
   515→  const [analyzeLine, setAnalyzeLine] = useState(0);
   516→  const [noteText, setNoteText] = useState("");
   517→  const analyzeMutation = useMutation({
   518→    mutationFn: () => apiSend(`/api/incidents/${incidentId}/analyze`, "POST"),
   519→    onSuccess: (data: unknown) => {
   520→      let msg = "AI analysis complete — BLUF report ready.";
   521→      if (data && typeof data === "object" && "message" in data) {
   522→        const m = (data as { message: unknown }).message;
   523→        if (typeof m === "string") msg = m;
   524→      }
   525→      toast.success("Analysis complete", { description: msg });
   526→      queryClient.invalidateQueries({ queryKey: ["incident", incidentId] });
   527→      queryClient.invalidateQueries({ queryKey: ["incidents"] });
   528→      queryClient.invalidateQueries({ queryKey: ["dashboard-summary"] });
   529→    },

   535→  useEffect(() => {

   543→

   550→        : variables.classification

   552→          : variables.analystNote
   553→            ? "Analyst note saved"
   554→            : "Incident updated";
   555→      toast.success(what, { description: "Analyst feedback recorded." });
   556→      queryClient.invalidateQueries({ queryKey: ["incident", incidentId] });
   557→      queryClient.invalidateQueries({ queryKey: ["incidents"] });
   558→      queryClient.invalidateQueries({ queryKey: ["dashboard-summary"] });
   559→      if (variables.analystNote) setNoteText("");
   560→    },
   561→    onError: (err: Error) => {
   562→      toast.error("Update failed", { description: err.message });
   563→    },
   564→  });
   565→
   566→  if (detailQuery.isLoading) {
   567→    return (
   568→      <div className="flex flex-col gap-4 sm:gap-6" aria-busy="true" aria-label="Loading incident detail">
   569→        <Skeleton className="h-36 rounded-xl" />
   570→        <Skeleton className="h-52 rounded-xl" />
   571→        <Skeleton className="h-64 rounded-xl" />
   572→      </div>
   573→    );
   574→  }
   575→  if (detailQuery.isError) {
   576→    return (
   577→      <ErrorState
   578→        message={detailQuery.error instanceof Error ? detailQuery.error.message : undefined}
   579→        onRetry={() => detailQuery.refetch()}
   580→      />
   581→    );
   582→  }
   583→  const incident = detailQuery.data;
   584→  if (!incident) return null;
   585→
   586→  return (
   587→    <div className="flex min-w-0 flex-col gap-4 sm:gap-6">
   588→      {/* Header */}
   589→      <motion.div
   590→        key={incident.id}
   591→        initial={{ opacity: 0, y: 10 }}
   592→        animate={{ opacity: 1, y: 0 }}
   593→        transition={{ duration: 0.35 }}
   594→      >
   595→        <Card className="rounded-xl p-4 sm:p-6">
   596→          <div className="flex flex-col gap-4">

   602→                {incident.incidentId}
   603→              </span>
   604→            </div>
   605→            <div>
   606→              <h2 className="text-lg font-bold leading-snug tracking-tight sm:text-xl">
   607→                {incident.title}
   608→              </h2>
   609→              <p className="mt-1 text-xs text-muted-foreground">
   610→                {incident.alertCount} related alert{incident.alertCount === 1 ? "" : "s"}
   611→                {" · "}
   612→                {incident.sources.join(", ") || "single source"}
   613→                {" · "}
   614→                created {timeAgo(incident.createdAt)}
   615→              </p>
   616→            </div>
   617→
   618→            {/* Actions */}
   619→            <div className="flex flex-wrap items-center gap-2 border-t border-border/70 pt-4">
   620→              <Button
   621→                type="button"
   622→                onClick={() => {
   623→                  setAnalyzeLine(0);
   624→                  analyzeMutation.mutate();
   625→                }}
   626→                disabled={analyzeMutation.isPending}
   627→                className="min-h-11 gap-2 border-emerald-500/50 bg-emerald-500/15 font-semibold text-emerald-300 hover:bg-emerald-500/25 hover:text-emerald-200"
   628→                aria-label="Run AI analysis on this incident"
   629→              >
   630→                {analyzeMutation.isPending ? (
   631→                  <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   632→                ) : (
   633→                  <Sparkles className="size-4" aria-hidden="true" />
   634→                )}
   635→                {analyzeMutation.isPending ? (
   636→                  <span className="font-mono text-xs">{ANALYZE_LINES[analyzeLine]}</span>
   637→                ) : (
   638→                  "Run AI Analysis"
   639→                )}
   640→              </Button>
   641→
   642→              <div className="flex items-center gap-2">
   643→                <label htmlFor="workflow-status" className="sr-only">Incident status</label>
   644→                <Select
   645→                  value={incident.status}
   646→                  onValueChange={(v) =>
   647→                    updateMutation.mutate({ status: v as IncidentStatus })
   648→                  }
   649→                  disabled={updateMutation.isPending}
   650→                >
   651→                  <SelectTrigger id="workflow-status" className="min-h-11 w-40" aria-label="Set incident status">
   652→                    <SelectValue placeholder="Status" />
   653→                  </SelectTrigger>
   654→                  <SelectContent>
   655→                    {STATUSES.map((s) => (
   656→                      <SelectItem key={s} value={s}>
   657→                        {s}
   658→                      </SelectItem>
   659→                    ))}
   660→                  </SelectContent>
   661→                </Select>
   662→              </div>
   663→
   664→              <div className="ml-auto flex items-center gap-1.5">
   665→                <Button
   666→                  type="button"
   667→                  variant="outline"
   668→                  size="sm"
   669→                  className="min-h-11 gap-1.5 border-emerald-500/40 text-emerald-300 hover:bg-emerald-500/10 hover:text-emerald-200"
   670→                  disabled={updateMutation.isPending}
   671→                  onClick={() =>
   672→                    updateMutation.mutate({ classification: "Genuine Threat" as Classification })
   673→                  }
   674→                  aria-label="Confirm this incident as a genuine threat"
   675→                >
   676→                  <ThumbsUp className="size-3.5" aria-hidden="true" />
   677→                  Confirm Threat
   678→                </Button>
   679→                <Button
   680→                  type="button"
   681→                  variant="outline"
   682→                  size="sm"
   683→                  className="min-h-11 gap-1.5 border-slate-500/40 text-slate-300 hover:bg-slate-500/10 hover:text-slate-200"

   685→                  onClick={() =>
   686→                    updateMutation.mutate({ classification: "False Positive" as Classification })

   688→                  aria-label="Mark this incident as a false positive"
   689→                >
   690→                  <ThumbsDown className="size-3.5" aria-hidden="true" />
   691→                  Mark False Positive
   692→                </Button>

   694→                  type="button"
   695→                  variant="ghost"
   696→                  size="icon"
   697→                  className="size-11 shrink-0 text-muted-foreground"

   701→                  <X className="size-4" aria-hidden="true" />
   702→                </Button>
   703→              </div>
   704→            </div>
   705→
   706→            {/* Analyst note */}
   707→            <div className="flex flex-col gap-2 border-t border-border/70 pt-4">

   710→                className="flex items-center gap-2 text-[11px] font-semibold uppercase tracking-wider text-muted-foreground"

   712→                <NotebookPen className="size-3.5 text-emerald-400" aria-hidden="true" />
   713→                Analyst Working Notes
   714→              </label>
   715→              <div className="flex flex-col gap-2 sm:flex-row">

   717→                  id="analyst-note"
   718→                  value={noteText}

   756→              </div>
   757→              <p className="font-mono text-[10px] text-muted-foreground/70">
   758→                {noteText.length}/600 characters · notes are timestamped and appended to the case file

   761→          </div>
   762→        </Card>

   764→
   765→      {/* Unanalyzed ribbon */}

   772→          Deterministic scoring only — run AI Analysis for the full LLM investigation report.
   773→        </div>
   774→      )}
   775→
   776→      <ScorePanel incident={incident} />
   777→
   778→      <div className="grid grid-cols-1 gap-4 sm:gap-6 xl:grid-cols-2">
   779→        <AttackTimeline incident={incident} />
   780→        <div className="flex flex-col gap-4 sm:gap-6">
   781→          <MitrePanel incident={incident} />
   782→          {incident.explanation && (
   783→            <Card className="rounded-xl">
   784→              <CardHeader className="pb-3">
   785→                <CardTitle className="flex items-center gap-2 text-sm font-semibold uppercase tracking-wider text-muted-foreground">

   792→                  {incident.explanation}
   793→                </p>
   794→              </CardContent>
   795→            </Card>
   796→          )}
   797→        </div>
   798→      </div>
   799→
   800→      <BlufReport incident={incident} />
   801→    </div>
   802→  );
   803→}

   104→   104→function buildAlertsUrl(filters: FeedFilters): string {
   105→   105→  const params = new URLSearchParams();
   106→   106→  if (filters.search.trim()) params.set("search", filters.search.trim());
   107→   107→  if (filters.source !== "all") params.set("source", filters.source);
   108→   108→  if (filters.severity !== "all") params.set("severity", filters.severity);
   109→   109→  if (filters.correlated === "correlated") params.set("correlated", "true");
   110→   110→  if (filters.hideAck) params.set("ack", "unack");
   111→   111→  const qs = params.toString();
   112→   112→  return qs ? `/api/alerts?${qs}` : "/api/alerts";
   113→   113→}

   125→   125→const PRESETS_KEY = "sentinelai.feedPresets";
   126→   126→
   127→   127→function loadPresets(): FeedPreset[] {
   128→   128→  if (typeof window === "undefined") return []; // SSR guard
   129→   129→  try {
   130→   130→    const raw = localStorage.getItem(PRESETS_KEY);
   131→   131→    if (!raw) return [];
   132→   132→    const parsed = JSON.parse(raw) as unknown;
   133→   133→    if (!Array.isArray(parsed)) return [];
   134→   134→    return parsed.filter(
   135→   135→      (p): p is FeedPreset =>
   136→   136→        typeof p === "object" && p !== null &&
   137→   137→        typeof (p as FeedPreset).id === "string" &&
   138→   138→        typeof (p as FeedPreset).name === "string" &&
   139→   139→        typeof (p as FeedPreset).filters === "object"
   140→   140→    );
   141→   141→  } catch {
   142→   142→    return [];
   143→   143→  }
   144→   144→}
   145→   145→
   146→   146→function savePresets(presets: FeedPreset[]) {
   147→   147→  try {
   148→   148→    localStorage.setItem(PRESETS_KEY, JSON.stringify(presets));
   149→   149→  } catch {
   150→   150→    // storage full/blocked — presets just won't persist
   151→   151→  }
   152→   152→}

   154→   154→function filtersMatch(a: FeedFilters, b: FeedFilters): boolean {
   155→   155→  return (
   156→   156→    a.search === b.search &&
   157→   157→    a.source === b.source &&
   158→   158→    a.severity === b.severity &&
   159→   159→    a.correlated === b.correlated &&
   160→   160→    a.hideAck === b.hideAck
   161→   161→  );
   162→   162→}

   170→   170→function guessFormatFromName(name: string): NonNullable<ImportPayload["format"]> | null {
   171→   171→  const ext = name.toLowerCase().split(".").pop() ?? "";
   172→   172→  if (ext === "json") return "json";
   173→   173→  if (ext === "csv" || ext === "tsv") return "csv";
   174→   174→  if (ext === "txt" || ext === "log" || ext === "text") return "text";
   175→   175→  return null;
   176→   176→}
